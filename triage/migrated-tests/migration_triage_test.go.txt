 "encoding/json"

 "os/exec"
 "path/filepath"
 "regexp"
 "testing"

 "github.com/ballerina-nutcracker/ballerina/test_util"
 _ "github.com/ballerina-nutcracker/ballerina/lib/rt"
)

func TestMigrationTriage(t *testing.T) {
 var input struct { Name string; Path string; Project bool; Args []string }
 if err := json.Unmarshal([]byte(os.Getenv("MIGRATION_TRIAGE_CASE")), &input); err != nil { t.Fatal(err) }
 tc := test_util.TestCase{Name: input.Name, InputPath: input.Path, IsProject: input.Project}
 p := NewTestPal()
 defer func() {
  data, _ := json.Marshal(struct {Stdout string; Stderr string; Diagnostics []ResolvedDiag}{p.Stdout(), p.Stderr(), p.Diagnostics()})
  _ = os.WriteFile(os.Getenv("MIGRATION_TRIAGE_CAPTURE"), data, 0600)
 }()
 if input.Args != nil {
  argv := append([]string{"run", input.Path, "--"}, input.Args...)
  cmd := exec.Command(os.Getenv("MIGRATION_TRIAGE_BAL"), argv...)
  cmd.Dir = filepath.Dir(input.Path)
  var stdout, stderr captureBuffer
  cmd.Stdout, cmd.Stderr = &stdout, &stderr
  err := cmd.Run()
  p.WriteStdout(string(stdout)); p.WriteStderr(string(stderr))
  if tc.Suffix() == test_util.SuffixValid && err != nil { t.Errorf("CLI exit: %v", err) }
  if tc.Suffix() == test_util.SuffixPanic && err == nil { t.Error("expected nonzero exit") }
 } else {
  Run(t, tc, p, nil)
 }
 if regexp.MustCompile(`(?m)^fatal\[`).MatchString(p.Stderr()) { t.Errorf("compilation produced a fatal diagnostic; this is not a successful compile or expected runtime panic") }
 if tc.Suffix() == test_util.SuffixValid || tc.Suffix() == test_util.SuffixPanic {
  if regexp.MustCompile(`(?m)^error\[`).MatchString(p.Stderr()) { t.Errorf("valid/panic case did not compile successfully") }
 }
 checkExpectedOutputInvariants(t, tc, p.Stdout(), normalizeIntegrationStderr(p.Stderr()))
 assertAnnotations(t, tc, p.Stdout(), p.Stderr(), p.Diagnostics())
}

type captureBuffer []byte
func (b *captureBuffer) Write(p []byte) (int,error) { *b = append(*b,p...); return len(p),nil }
